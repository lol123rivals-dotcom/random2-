local function GetFruitOriginalName(tool)
    if not tool then return nil end
    local orig = tool:GetAttribute("OriginalName")
    if orig and orig ~= "" then return orig end
    if FruitOriginalNames[tool.Name] then
        return FruitOriginalNames[tool.Name]
    end
    local clean = string.gsub(tool.Name, "%s*Fruit%s*", "")
    clean = string.gsub(clean, "^%s*(.-)%s*$", "%1")
    return clean .. "-" .. clean
end

-- 5. CENTUDOX LINEAR TWEEN FLIGHT ENGINE
-- ------------------------------------------------------------
local function to(Pos)
    pcall(function()
        local char = LocalPlayer.Character
        if not char or not char:FindFirstChild("HumanoidRootPart") or not char:FindFirstChild("Humanoid") then
            return
        end
        if char.Humanoid.Health <= 0 then return end

        local hrp = char.HumanoidRootPart
        local Distance = (Pos.Position - hrp.Position).Magnitude

        if not hrp:FindFirstChild("Hold") then
            local Hold = Instance.new("BodyVelocity", hrp)
            Hold.Name = "Hold"
            Hold.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            Hold.Velocity = Vector3.new(0, 0, 0)
        end

        if char.Humanoid.Sit == true then
            char.Humanoid.Sit = false
        end

        local Speed = _G.FlySpeed or 350
        if Distance <= 15 then
            if tween then tween:Cancel() end
            hrp.CFrame = Pos
            return
        elseif Distance < 1000 then
            Speed = math.max(Speed, 375)
        else
            Speed = _G.FlySpeed or 350
        end

        pcall(function()
            if tween then tween:Cancel() end
            tween = TweenService:Create(
                hrp,
                TweenInfo.new(Distance / Speed, Enum.EasingStyle.Linear),
                { CFrame = Pos }
            )
            tween:Play()
        end)
    end)
end

local function stopTween()
    if tween then
        pcall(function() tween:Cancel() end)
        tween = nil
    end
    pcall(function()
        local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            local hold = hrp:FindFirstChild("Hold")
            if hold then hold.Velocity = Vector3.new(0, 0, 0) end
            hrp.Velocity = Vector3.new(0, 0, 0)
        end
    end)
end

local fruitESPMap = {}

local function UpdateFruitESP()
    pcall(function()
        local char = LocalPlayer.Character
        local head = char and char:FindFirstChild("Head")

        for _, v in ipairs(Workspace:GetChildren()) do
            if v:IsA("Tool") and (string.find(string.lower(v.Name), "fruit") or v.ToolTip == "Blox Fruit") then
                local handle = v:FindFirstChild("Handle") or v:FindFirstChildWhichIsA("BasePart")
                if handle then
                    local espName = "FruitESP_" .. sessionID
                    local bill = handle:FindFirstChild(espName)

                    if _G.FruitESP then
                        local dist = head and math.floor((head.Position - handle.Position).Magnitude) or 0
                        if not bill then
                            bill = Instance.new("BillboardGui")
                            bill.Name = espName
                            bill.Size = UDim2.new(0, 180, 0, 45)
                            bill.ExtentsOffset = Vector3.new(0, 2, 0)
                            bill.AlwaysOnTop = true
                            bill.Adornee = handle
                            bill.Parent = handle

                            local text = Instance.new("TextLabel")
                            text.Name = "ESPLabel"
                            text.Size = UDim2.new(1, 0, 1, 0)
                            text.BackgroundTransparency = 1
                            text.TextSize = 14
                            text.Font = Enum.Font.GothamBold
                            text.TextColor3 = Color3.fromRGB(255, 85, 85)
                            text.TextStrokeTransparency = 0
                            text.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                            text.Parent = bill
                            fruitESPMap[handle] = bill
                        end
                        local lbl = bill:FindFirstChild("ESPLabel")
                        if lbl then
                            lbl.Text = v.Name .. "\n[" .. dist .. "m]"
                        end
                    else
                        if bill then
                            bill:Destroy()
                            fruitESPMap[handle] = nil
                        end
                    end
                end
            end
        end
    end)
end

task.spawn(function()

        UpdateFruitESP()
    end

-- 8. SPAWNED FRUIT DETECTION
-- ------------------------------------------------------------
local function GetSpawnedFruits()
    local list = {}
    local function checkItem(v)
        if v:IsA("Tool") and (string.find(string.lower(v.Name), "fruit") or v.ToolTip == "Blox Fruit") then
            local handle = v:FindFirstChild("Handle") or v:FindFirstChildWhichIsA("BasePart")
            if handle then
                table.insert(list, {
                    Object   = v,
                    Handle   = handle,
                    Name     = v.Name,
                    Position = handle.Position
                })
            end
        end
    end

    for _, v in ipairs(Workspace:GetChildren()) do
        checkItem(v)
    end

    local wo = Workspace:FindFirstChild("_WorldOrigin")
    if wo then
        for _, v in ipairs(wo:GetChildren()) do
            checkItem(v)
        end
    end

    return list
end

-- 9. AUTO STORE FRUITS
-- ------------------------------------------------------------
local function StoreAllFruits()
    pcall(function()
        local toolsToStore = {}

        if LocalPlayer.Character then
            for _, item in ipairs(LocalPlayer.Character:GetChildren()) do
                if item:IsA("Tool") and (string.find(string.lower(item.Name), "fruit") or item.ToolTip == "Blox Fruit") then
                    table.insert(toolsToStore, item)
                end
            end
        end

        local bp = LocalPlayer:FindFirstChild("Backpack")
        if bp then
            for _, item in ipairs(bp:GetChildren()) do
                if item:IsA("Tool") and (string.find(string.lower(item.Name), "fruit") or item.ToolTip == "Blox Fruit") then
                    table.insert(toolsToStore, item)
                end
            end
        end

        local remotes = ReplicatedStorage:FindFirstChild("Remotes")
        local commF = remotes and remotes:FindFirstChild("CommF_")
        if not commF then return end

        for _, tool in ipairs(toolsToStore) do
            local origName = GetFruitOriginalName(tool)
            if origName then
                pcall(function()
                    commF:InvokeServer("StoreFruit", origName, tool)
                end)
                task.wait(0.3)
            end
        end
    end)
end

-- 10. EXACT SERVER HOP ENGINE (FROM AUTO BOUNTY, NO COMBAT CHECK)
-- ------------------------------------------------------------
function HopServer(counts)
    if counts == nil then
        counts = 8
    end
    isHopping = true
    stopTween()
    currentStatusText = "Hopping Server..."

    local function dismissPrompt(v)
        if v.Name == "ErrorPrompt" then
            if v.Visible then
                if v:FindFirstChild("TitleFrame") and v.TitleFrame:FindFirstChild("ErrorTitle") and v.TitleFrame.ErrorTitle.Text == "Teleport Failed" then
                    v.Visible = false
                end
            end
            v:GetPropertyChangedSignal("Visible"):Connect(
                function()
                    if v.Visible then
                        if v:FindFirstChild("TitleFrame") and v.TitleFrame:FindFirstChild("ErrorTitle") and v.TitleFrame.ErrorTitle.Text == "Teleport Failed" then
                            v.Visible = false
                        end
                    end
                end
            )
        end
    end

    pcall(function()
        for i, v in game.CoreGui.RobloxPromptGui.promptOverlay:GetChildren() do
            dismissPrompt(v)
        end
        game.CoreGui.RobloxPromptGui.promptOverlay.ChildAdded:Connect(dismissPrompt)
    end)

    pcall(function()
        game:GetService("Players").LocalPlayer.PlayerGui.ServerBrowser.Frame.Filters.SearchRegion.TextBox.Text = "Singapore"
    end)

    while wait() do
        if not _G.AutoCollectHop then
            isHopping = false
            break
        end
        for r = 1, math.huge do
            if not _G.AutoCollectHop then
                isHopping = false
                break
            end
            local s, servers = pcall(function()
                return game.ReplicatedStorage.__ServerBrowser:InvokeServer(r)
            end)
            if s and type(servers) == "table" then
                for k, v in pairs(servers) do
                    if not _G.AutoCollectHop then
                        isHopping = false
                        break
                    end
                    if k ~= game.JobId and type(v) == "table" and v["Count"] and v["Count"] <= counts then
                        currentStatusText = "Teleporting (" .. tostring(v["Count"]) .. " players)..."
                        game.ReplicatedStorage.__ServerBrowser:InvokeServer("teleport", k)
                        task.wait(2)
                    end
                end
            else
                if not _G.AutoCollectHop then
                    isHopping = false
                    break
                end
                currentStatusText = "Teleporting via TeleportService..."
                pcall(function()
                    game:GetService("TeleportService"):Teleport(game.PlaceId)
                end)
                break
            end
        end
    end
    isHopping = false
end
getgenv().HopServer = HopServer

-- 11. FRUIT COLLECTION LOOP (RESTARTABLE ON PLAY)
-- ------------------------------------------------------------
local function StartFruitCollector()
    if collectorThread then
        task.cancel(collectorThread)
        collectorThread = nil
    end

    collectorThread = task.spawn(function()
        while _G.AutoCollectHop do
            if not isHopping then
                local fruits = GetSpawnedFruits()

                if #fruits > 0 then
                    currentStatusText = string.format("Found %d fruit(s)! Grabbing...", #fruits)

                    while #fruits > 0 and _G.AutoCollectHop and not isHopping do
                        local char = LocalPlayer.Character
                        local myHRP = char and char:FindFirstChild("HumanoidRootPart")

                        if myHRP then
                            table.sort(fruits, function(a, b)
                                return (a.Position - myHRP.Position).Magnitude < (b.Position - myHRP.Position).Magnitude
                            end)

                            local fruitData = fruits[1]
                            activeFruitTarget = fruitData.Name
                            currentStatusText = "Flying to: " .. fruitData.Name
                            local targetPos = CFrame.new(fruitData.Position + Vector3.new(0, 1.5, 0))

                            local timeout = 0
                            local collected = false

                            while _G.AutoCollectHop and not isHopping and timeout < 25 do
                                task.wait(0.1)
                                timeout = timeout + 0.1

                                local myChar = LocalPlayer.Character
                                local hrpNow = myChar and myChar:FindFirstChild("HumanoidRootPart")
                                if not hrpNow then break end

                                if not fruitData.Object or fruitData.Object.Parent ~= Workspace then
                                    collected = true
                                    break
                                end

                                local dist = (fruitData.Position - hrpNow.Position).Magnitude
                                if dist <= 12 then
                                    pcall(function()
                                        hrpNow.CFrame = CFrame.new(fruitData.Position)
                                        local hum = myChar:FindFirstChildOfClass("Humanoid")
                                        if hum and fruitData.Object and fruitData.Object.Parent == Workspace then
                                            hum:EquipTool(fruitData.Object)
                                        end
                                    end)
                                    task.wait(0.4)
                                    collected = true
                                    break
                                else
                                    to(targetPos)
                                end
                            end

                            stopTween()

                            if collected or (fruitData.Object and fruitData.Object.Parent ~= Workspace) then
                                currentStatusText = "Collected: " .. fruitData.Name .. "!"
                                _G.TotalFruitsCollected = (_G.TotalFruitsCollected or 0) + 1

                                if _G.AutoStoreFruit then
                                    currentStatusText = "Storing: " .. fruitData.Name .. "..."
                                    task.wait(0.3)
                                    StoreAllFruits()
                                    task.wait(0.3)
                                end
                            end
                        end

                        fruits = GetSpawnedFruits()
                    end
                else
                    -- No fruits found -> Hopping immediately using HopServer()
                    if _G.AutoCollectHop and not isHopping then
                        isHopping = true
                        currentStatusText = "No fruits found. Hopping server..."
                        task.wait(1)
                        if _G.AutoCollectHop then
                            HopServer()
                        else
                            isHopping = false
                        end
                    end
                end
            end
            task.wait(0.5)
        end
    end)
end

-- ------------------------------------------------------------
-- 12. DRAG HELPER (PC & MOBILE COMPATIBLE)
-- ------------------------------------------------------------
local function makeDraggable(trigger, target)
    local dragging = false
    local dragStart, startPos

    trigger.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = target.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            target.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
end

local function updateButtonVisual(running)
    if running then
        TextButton_6335.Text = "STOP"
        btnStroke.Color = Color3.fromRGB(220, 50, 50)
        statusLabel.TextColor3 = Color3.fromRGB(255, 220, 110)
    else
        TextButton_6335.Text = "PLAY"
        btnStroke.Color = Color3.fromRGB(50, 200, 90)
        statusLabel.TextColor3 = Color3.fromRGB(180, 180, 190)
    end
end

-- Stop / Play button handler:
-- STOP: immediately cancels flight/tween, cancels server hop, stops script completely.
-- PLAY: resumes everything from scratch.
TextButton_6335.MouseButton1Click:Connect(function()
    _G.AutoCollectHop = not _G.AutoCollectHop
    if _G.AutoCollectHop then
        isHopping = false
        currentStatusText = "Fruit Hunter resumed. Searching fruits..."
        updateButtonVisual(true)
        StartFruitCollector()
    else
        isHopping = false
        stopTween()
        currentStatusText = "Script stopped (Paused)."
        updateButtonVisual(false)
    end
end)

-- Button click tactile feedback
TextButton_6335.MouseButton1Down:Connect(function()
    TextButton_6335.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    TextButton_6335.TextColor3 = Color3.fromRGB(12, 12, 15)
end)
local function resetButtonVisual()
    TweenService:Create(TextButton_6335, TweenInfo.new(0.18), {
        BackgroundColor3 = Color3.fromRGB(0, 0, 0),
        TextColor3 = Color3.fromRGB(255, 255, 255)
    }):Play()
end
TextButton_6335.MouseButton1Up:Connect(resetButtonVisual)
TextButton_6335.MouseLeave:Connect(resetButtonVisual)

-- Minimize button -> Hides Main frame, shows top-left toggle
Minimze.MouseButton1Click:Connect(function()
    Frame_4273.Visible = false
    MinToggle.Visible = true
end)

-- Minimized toggle button -> Restores Main frame
MinToggle.MouseButton1Click:Connect(function()
    MinToggle.Visible = false
    Frame_4273.Visible = true
end)

    while task.wait(0.5) do

            statusLabel.Text = "Status: " .. currentStatusText
            local fruits = GetSpawnedFruits()
            fruitsCountLabel.Text = string.format("Spawned Fruits: %d | Total: %d", #fruits, _G.TotalFruitsCollected or 0)
            targetFruitLabel.Text = "Target Fruit: " .. tostring(activeFruitTarget or "None")
        end)
    end
end)
